create database db_pedidos;
create table usuarios(
id_usu int primary key not null auto_increment,
nome_usu varchar(255) not null,
email varchar(255) not null,
telefone int (11) not null,
tipo int  not null);
create table pedidos(
id_pedidos int primary key not null auto_increment,
endereco_origem varchar(255) not null,
endereco_entrega varchar(255) not null,
valor_entrega decimal(10,2) not null,
status varchar(50) not null,
client_id int not null,
entregador_id int not null,
foreign key (client_id) references usuarios(id_usu),
foreign key (entregador_id) references usuarios(id_usu)
);
create table historico_status(
id_status int primary key not null auto_increment,
pedido_id int not null,
status varchar(50) not null,
data_alteracao TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
foreign key (pedido_id) references pedidos(id_pedidos)
);select * from usuarios
